Calculate a Hash Index

Calculate a hash index by adding the character codes of a key and using the table size.

const tableSize = 10;

// Calculate the hash index
Answer 
const key = "CAT";
const tableSize = 10;

let hash = 0;

for (let i = 0; i < key.length; i++) {
    hash += key.charCodeAt(i);
}

const index = hash % tableSize;

console.log(index);

// Output:
4

Example 19

Count Binary Tree Leaf Nodes

Count the leaf nodes in the following binary tree. 

const tree = {
    value: 1,
    left: {
        value: 2,
        left: null,
        right: null
    },
    right: {
        value: 3,
        left: null,
        right: null
    }
};

// Count leaf nodes
Answer 
const tree = {
    value: 1,
    left: {
        value: 2,
        left: null,
        right: null
    },
    right: {
        value: 3,
        left: null,
        right: null
    }
};

function countLeaves(node) {
    if (node === null) {
        return 0;
    }

    if (node.left === null && node.right === null) {
        return 1;
    }

    return countLeaves(node.left) +
        countLeaves(node.right);
}

console.log(countLeaves(tree));

// Output:
2
Example 20

Search a Binary Search Tree

Search for the value 7 in a binary search tree. 

const bst = {
    value: 8,
    left: {
        value: 3,
        left: null,
        right: null
    },
    right: {
        value: 10,
        left: null,
        right: null
    }
};

// Search for 7
Answer 
const bst = {
    value: 8,
    left: {
        value: 3,
        left: null,
        right: null
    },
    right: {
        value: 10,
        left: null,
        right: null
    }
};

function searchBST(node, target) {
    if (node === null) {
        return false;
    }

    if (node.value === target) {
        return true;
    }

    if (target < node.value) {
        return searchBST(node.left, target);
    }

    return searchBST(node.right, target);
}

console.log(searchBST(bst, 7));

// Output:
false

Example 22

Traverse a Graph Using BFS

Traverse the graph starting from node A using breadth-first search. 

const graph = {
    A: ["B", "C"],
    B: ["A", "D"],
    C: ["A"],
    D: ["B"]
};

// Perform BFS from A
Answer 
const graph = {
    A: ["B", "C"],
    B: ["A", "D"],
    C: ["A"],
    D: ["B"]
};

function bfs(graph, start) {
    const visited = new Set([start]);
    const queue = [start];

    while (queue.length > 0) {
        const node = queue.shift();

        console.log(node);

        for (const neighbor of graph[node]) {
            if (!visited.has(neighbor)) {
                visited.add(neighbor);
                queue.push(neighbor);
            }
        }
    }
}

bfs(graph, "A");

// Output:
A
B
C
D
